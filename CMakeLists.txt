set(ASTERION_SANITIZERS "" CACHE STRING "Optional address,undefined or thread sanitizer")
if(ASTERION_SANITIZERS)
  if(NOT ASTERION_SANITIZERS MATCHES "^(address,undefined|thread)$")
    message(FATAL_ERROR "Sanitizers require Clang/GCC and address,undefined or thread")
  endif()
  # -g1 keeps line tables for sanitizer stacks; full Linux DWARF would push
  # instrumented services past the Agent's 128 MiB artifact limit.
  target_compile_options(asterion_foundation PUBLIC -fsanitize=${ASTERION_SANITIZERS} -fno-omit-frame-pointer -g1)
  target_link_options(asterion_foundation PUBLIC -fsanitize=${ASTERION_SANITIZERS})
  # Instrumented builds run several times slower; timing assertions scale.
  target_compile_definitions(asterion_foundation PUBLIC ASTERION_SANITIZED=1)
endif()

  # Optional arguments: a timeout in seconds, then SERIAL for tests whose
  # timing depends on the machine not being shared with other tests.
  function(asterion_unit_tests target source library)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE ${library} GTest::gtest_main)
    set(test_timeout 30)
    if(ARGC GREATER 3)
      set(test_timeout ${ARGV3})
    endif()
    if(ASTERION_SANITIZERS)
      math(EXPR test_timeout "${test_timeout} * 5")
    endif()
    set(run_serial FALSE)
    if(ARGC GREATER 4 AND "${ARGV4}" STREQUAL "SERIAL")
      set(run_serial TRUE)
    endif()
    # Instrumented binaries can need more than CMake's five-second default to
    # start while other targets run. This changes discovery, not case budgets.
    gtest_discover_tests(${target} DISCOVERY_MODE PRE_TEST DISCOVERY_TIMEOUT 30
      PROPERTIES TIMEOUT ${test_timeout} RUN_SERIAL ${run_serial}
        ENVIRONMENT_MODIFICATION "${asterion_test_node_capacity}")
  endfunction()
